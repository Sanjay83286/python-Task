a="aabcde"
list1=[]
dic1={}
for i in range(1,len(a)+1):
    sub=''
    for j in range(i):
        sub=sub+a[j]
    print("sam",sub)
    reverse=sub[::-1]
    if reverse==sub:
        #list creation and appending
        list1.append(reverse)
        #dictionary creation and appending
        dic1[reverse]=len(reverse)
print(list1,dic1)
